def binary_search(arr, target):
    low, high = 0, len(arr) - 1
    while low <= high:
        mid = (low + high) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return -1

accs = sorted([int(x) for x in input("Enter IDs: ").split()])
print(f"Sorted: {accs}")
target = int(input("Search ID: "))

idx = binary_search(accs, target)
print(f"Found at index: {idx}" if idx != -1 else "Not found")
