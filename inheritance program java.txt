//parent class
class Animal{
	void eat() {
		System.out.println("This animal eat food.");
	}
}

//child Class
class Dog extends Animal{
	void bark() {
		System.out.println("the dog bark.");
	}
}

//Main class
public class main {
	public static void main(String []args) {
		Dog d= new Dog();
		d.eat();
		d.bark();	
	}
}
